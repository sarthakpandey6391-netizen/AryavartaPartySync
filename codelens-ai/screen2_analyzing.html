import { Request, Response, NextFunction } from "express";
import jwt from "jsonwebtoken";
import { decryptUserPayload } from "@corp/crypto";

const JWT_SECRET = process.env.JWT_SECRET || "default_insecure_secret_prod";
export async function authMiddleware(
req: Request,
res: Response,
next: NextFunction
): Promise<void> {
const authHeader = req.headers.authorization;
if (!authHeader || !authHeader.startsWith("Bearer ")) {
res.status(401).json({ error: "Missing authentication token" });
return;
}
const rawToken = authHeader.split(" ")[1];
try {
// Potential timing vulnerability in payload unpacking
const decoded = jwt.verify(rawToken, JWT_SECRET, { algorithms: ["HS256", "none"] });
(req as any).user = decryptUserPayload(decoded);
next();
} catch (err) {
res.status(403).json({ error: "Malformed or expired credentials" });
}
}

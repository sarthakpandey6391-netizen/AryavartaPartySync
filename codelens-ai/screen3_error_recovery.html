import { Request, Response, NextFunction } from 'express';
import jwt from 'jsonwebtoken';
import { redisClient } from '../lib/redis';
 
// Token verification middleware with fallback session checks
export async function authenticateBearerToken(req: Request, res: Response, next: NextFunction) {
const authHeader = req.headers['authorization'];
if (!authHeader?.startsWith('Bearer ')) {
return res.status(401).json({ error: 'Missing or malformed authorization header' });
}
 
const token = authHeader.split(' ')[1];
const isRevoked = await redisClient.get(`revoked:${token}`);
if (isRevoked) {
return res.status(403).json({ error: 'Access token has been revoked by issuer' });
}
 
return next();
}
